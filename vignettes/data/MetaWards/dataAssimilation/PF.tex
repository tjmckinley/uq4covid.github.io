\section{Auxiliary Particle Filter (3)}

\subsection{Likelihood terms}

Let's first consider the form of the likelihood functions for the simulator and the model discrepancy terms. We let $\bz_t = \left(D^{Z^\prime}_{Ht}, D^{Z^\prime}_{It}\right)$ be the \textbf{observed} death incidence; $\bx_t = \left(E^X_t, A^X_t, R^X_{At}, P^X_t, I^X_{1t}, I^X_{2t}, R^X_{It}, D^X_{It}, H^X_T, D^X_{Ht}, R^X_{Ht}\right)$ be a vector of counts from the \textbf{simulator} at time $t$; and $\by_t = \left(E^Y_t, A^Y_t, R^Y_{At}, P^Y_t, I^Y_{1t}, I^Y_{2t}, R^Y_{It}, D^Y_{It}, H^Y_T, D^Y_{Ht}, R^Y_{Ht}\right)$ be a vector of counts \textbf{adjusted for model discrepancy} at time $t$. Let $\bA_t = \left\{\by^j, \bx^j_t, \pi^j_t; j = 1, \dots, M\right\}$ denote the swarm of particles at time $t$, where $\pi^j_t$ denotes a weight for particle $j$, such that $\sum_{j = 1}^M \pi^j_t = 1$.

We note that some of the likelihood terms below depend on \textbf{incidence} counts (e.g. $D^{X^\prime}_{Ht}$), but these can be directly derived from the raw counts if required. In the discussion below we use $\pi\left(\cdot\right)$ to denote an arbitrary probability mass function, and we drop dependence on the parameters, $\btheta$, for brevity. \textcolor{red}{We note that in all p.m.f. functions below we product over the individual age-classes and all age-classes are independent, and so can be sampled separately using the approaches below, with the exception of the transmission process, but this is handled in the simulator. I've ignored age below to try to simplify...}

Hence at time $t$ the \textbf{empirical prediction density} defined by the particles is:
\begin{equation}
    \hat{\pi}\left(\by_t, \bx_t \mid \bz_{1:(t - 1)}\right) = \sum_{j = 1}^M \pi\left(\by_t \mid \bx_t, \by^j_{t - 1}\right)\pi\left(\bx_t \mid \by^j_{t - 1}\right)\pi^j_{t - 1}.
\end{equation}
The \textbf{empirical filtering distribution} is:
\begin{equation}
    \hat{\pi}\left(\by_t, \bx_t \mid \bz_{1:t}\right) \propto \pi\left(\bz_t \mid \by_t\right)\sum_{j = 1}^M \pi\left(\by_t \mid \bx_t, \by^j_{t - 1}\right)\pi\left(\bx_t \mid \by^j_{t - 1}\right)\pi^j_{t - 1}.
\end{equation}
Here the simulator distribution is:
\begin{align}
    \pi\left(\bx_t \mid \by^j_{t - 1}\right) &= \pi\left(D^{X^\prime}_{Ht}, R^{X^\prime}_{Ht} \mid H^{Y^j}_{t - 1}\right)\pi\left(H^{X^\prime}_t, I^{X^\prime}_{2t}, D^{X^\prime}_{It} \mid I^{Y^j}_{1(t - 1)}\right)\pi\left(R^{X^\prime}_{It} \mid I^{Y^j}_{2(t - 1)}\right)\nonumber\\
    & \qquad \qquad \qquad \times \pi\left(I^{X^\prime}_{1t} \mid P^{Y^j}_{t - 1}\right)\pi\left(P^{X^\prime}_t, A^{X^\prime}_t \mid E^{Y^j}_{t - 1}\right)\pi\left(R^{X^\prime}_{At} \mid A^{Y^j}_{t - 1} \right)\nonumber\\
    & \qquad \qquad \qquad \qquad \qquad \qquad \times \pi\left(E^{X^\prime}_t \mid S^{Y^j}_{t - 1}, A^{Y^j}_{t - 1}, P^{Y^j}_{t - 1}, I^{Y^j}_{1(t - 1)}, I^{Y^j}_{2(t - 1)}\right),\label{eq:simlike}
\end{align}
and the conditional model discrepancy distribution is:
\begin{align}
    \pi\left(\by_t \mid \bx_t, \by^j_{t - 1}\right) &= \pi\left(D^{Y^\prime}_{Ht} \mid D^{X^\prime}_{Ht}, H^{Y^j}_{t - 1}\right)\pi\left(D^{Y^\prime}_{It} \mid D^{X^\prime}_{It}, I^{Y^j}_{1(t - 1)}\right)\pi\left(R^{Y^\prime}_{Ht} \mid R^{X^\prime}_{Ht}, D^{Y^\prime}_{Ht}, H^{Y^j}_{t - 1}\right)\nonumber\\
    & \qquad \times \pi\left(H^Y_t \mid H^X_t, H^{Y^j}_{t - 1}, D^{Y^\prime}_{Ht}, R^{Y^\prime}_{Ht}, D^{Y^\prime}_{It}, I^{Y^j}_{1(t - 1)}\right) \pi\left(R^{Y^\prime}_{It} \mid R^{X^\prime}_{It}, I^{Y^j}_{2(t - 1)}\right)\nonumber\\
    &\qquad \qquad \times \pi\left(I^Y_{2t} \mid I^X_{2t}, I^{Y^j}_{2(t - 1)}, R^{Y^\prime}_{It}, D^{Y^\prime}_{It}, H^{Y^\prime}_t, I^{Y^j}_{1(t - 1)}\right)\nonumber\\
    &\qquad \qquad \qquad \times\pi\left(I^Y_{1t} \mid I^X_{1t}, I^{Y^j}_{1(t - 1)}, I^{Y^\prime}_{2t}, D^{Y^\prime}_{It}, H^{Y^\prime}_t, P^{Y^j}_{t - 1}\right)\nonumber\\
    &\qquad \qquad \qquad \qquad \times \pi\left(P^Y_t \mid P^X_t, P^{Y^j}_{t - 1}, I^{Y^\prime}_{1t}, E^{Y^j}_{t - 1}\right)\pi\left(R^{Y^\prime}_{At} \mid R^{X^\prime}_{At}, A^{Y^j}_{t - 1}\right)\nonumber\\
    & \qquad \qquad \qquad \qquad \qquad \times \pi\left(A^Y_t \mid A^X_t, A^{Y^j}_{t - 1}, R^{Y^\prime}_{At}, E^{Y^j}_{t - 1}\right)\nonumber\\
    & \qquad \qquad \qquad \qquad \qquad \qquad \times \pi\left(E^Y_t \mid E^X_t, E^{Y^j}_{t - 1}, P^{Y^\prime}_t, A^{Y^\prime}_t, S^{Y^j}_{t - 1}\right).\label{eq:mdlike}
\end{align}
The conditional observation density is:
\begin{align}
    \pi\left(\bz_t \mid \by_t\right) &= \pi\left(D^{Z^\prime}_{Ht} \mid D^{Y^\prime}_{Ht}\right)\pi\left(D^{Z^\prime}_{It} \mid D^{Y^\prime}_{It}\right).\label{eq:obsdens}
\end{align}

\subsection{Importance sampling}

Now let's consider building an importance distribution where all of the unknowns are conditioned on the data. Following \cite{pitt_shephard:1999} we introduce an auxiliary variable $k$, denoting the particle index. In this case we can consider an importance distribution of the form:
\begin{equation}\label{eq:apf3}
    q\left(\by_t, \bx_t, k \mid \bz_t, \bA_{t - 1}\right) \propto \pi\left(\bz_t \mid \by_t\right)\pi\left(\by_t \mid \bx_t, \by^k_{t - 1}\right)\pi\left(\bx_t \mid \by^k_{t - 1}\right)\pi^k_{t - 1}.
\end{equation}

We can then sample from this conditionally, by considering the decomposition:
\begin{equation}\label{eq:qcond}
    q\left(\by_t, \bx_t, k \mid \bz_t, \bA_{t - 1}\right) = q\left(\by_t \mid \bx_t, k, \bz_t, \bA_{t - 1}\right)q\left(\bx_t \mid k, \bz_t, \bA_{t - 1}\right)q\left(k \mid \bz_t, \bA_{t - 1}\right),
\end{equation}
where
\begin{align}
    q\left(k \mid \bz_t, \bA_{t - 1}\right) &\propto \pi^k_{t - 1} \sum_{\bx_t}\left[ \pi\left(\bx_t \mid \by^k_{t - 1}\right) \sum_{\by_t} \pi\left(\bz_t \mid \by_t\right)\pi\left(\by_t \mid \bx_t, \by^k_{t - 1}\right)\right],\label{eq:sumxy}\\
    q\left(\bx_t \mid k, \bz_t, \bA_{t - 1}\right) &\propto \pi\left(\bx_t \mid \by^k_{t - 1}\right) \sum_{\by_t} \pi\left(\bz_t \mid \by_t\right)\pi\left(\by_t \mid \bx_t, \by^k_{t - 1}\right)\quad\mbox{and} \label{eq:sumy}\\
    q\left(\by_t \mid \bx_t, k, \bz_t, \bA_{t - 1}\right) &\propto \pi\left(\bz_t \mid \by_t\right)\pi\left(\by_t \mid \bx_t, \by^k_{t - 1}\right).\label{eq:sum}
\end{align}
These are derived by conditioning and marginalising out the relevant components in (\ref{eq:apf3}). In this case all unknowns, $k$, $\bx_t$ and $\by_t$ have finite (discrete) sample spaces, and hence the corresponding normalising constants can be calculated explicitly, and sampling can be done via inverse transform sampling.

Thus, an \textbf{auxiliary particle filter} can be conducted as follows.
\begin{enumerate}
    \item Set \textit{first-stage weights}:
    $$
        w^j_{t - 1 \mid t} = \pi^j_{t - 1} q\left(\bz_t \mid \by^j_{t - 1}\right) \qquad \mbox{and} \qquad \pi^j_{t - 1 \mid t} = \frac{w^j_{t - 1 \mid t}}{\sum_{l = 1}^M w^l_{t - 1 \mid t}}
    $$ 
    for $j = 1, \dots, M$, where
    $$
        q\left(\bz_t \mid \by^j_{t - 1}\right) = \sum_{\bx_t} \left[\pi\left(\bx_t \mid \by^j_{t - 1}\right)\sum_{\by_t} \pi\left(\bz_t \mid \by_t\right)\pi\left(\by_t \mid \bx_t, \by^j_{t - 1}\right)\right].
    $$
    \item Then, for $j = 1, \dots, M$:
    \item Sample $k^j \sim q\left(k \mid \bz_t, \bA_{t - 1}\right)$ where $q\left(k \mid \bz_t, \bA_{t - 1}\right) = \pi^k_{t - 1 \mid t}$.\label{apf:k}
    \item Sample $\bx^j_t \sim q\left(\bx_t \mid k^j, \bz_t, \bA_{t - 1}\right)$.\label{apf:x}
    \item Sample $\by^j_t \sim q\left(\by_t, \mid x^j_t, k^j, \bz_t, \bA_{t - 1}\right)$.\label{apf:y}
    \item Calculate the \textit{second-stage weights}:
    $$
        w^j_t = \frac{\pi\left(\bz_t \mid \by^j_t\right)\pi\left(\by^j_t \mid \bx^j_t, \by^{k^j}_{t - 1}\right)\pi\left(\bx^j_t \mid \by^{k^j}_{t - 1}\right)}{q\left(\bz_t \mid \by^{k^j}_{t - 1}\right)q\left(\by^j_t \mid \bx^j_t, k^j, \bz_t, \bA_{t - 1}\right)q\left(\bx^j_t \mid k^j, \bz_t, \bA_{t - 1}\right)} \qquad \mbox{and} \qquad \pi^j_t = \frac{w^j_t}{\sum_{l = 1}^M w^l_t}
    $$ 
    for $j = 1, \dots, M$.
    \item Calculate the likelihood contribution:
    $$
        \hat{\pi}\left(\bz_t \mid \bA_{t - 1}\right) = \left[\frac{1}{M}\sum_{j = 1}^M w^j_t\right]\left[\sum_{j = 1}^M w^j_{t - 1 \mid t}\right].
    $$
\end{enumerate}

\subsection{Conditional sampling}

If we first consider the sum in (\ref{eq:sumy}) we have that:
\begin{align}
    \sum_{\by_t} \pi\left(\bz_t \mid \by_t\right)\pi\left(\by_t \mid \bx_t, \by^k_{t - 1}\right) &= \sum_{D^{Y^\prime}_{Ht}}\sum_{D^{Y^\prime}_{It}}\dots\sum_{E^Y_t} \pi\left(\bz_t \mid \by_t\right)\pi\left(\by_t \mid \bx_t, \by^k_{t - 1}\right)\label{eq:margy}
\end{align}
where $\pi\left(\bz_t \mid \by_t\right)$ is given in (\ref{eq:obsdens}) and $\pi\left(\by_t \mid \bx_t, \by^k_{t - 1}\right)$ is given in (\ref{eq:mdlike}). Due to the conditioning, many of the components of (\ref{eq:mdlike}) marginalise to one in (\ref{eq:margy}), leaving:
\begin{align}
    \sum_{\by_t} \pi\left(\bz_t \mid \by_t\right)\pi\left(\by_t \mid \bx_t, \by^k_{t - 1}\right) &= \sum_{D^{Y^\prime}_{Ht}}\sum_{D^{Y^\prime}_{It}} \pi\left(D^{Z^\prime}_{Ht} \mid D^{Y^\prime}_{Ht}\right)\pi\left(D^{Z^\prime}_{It} \mid D^{Y^\prime}_{It}\right) \nonumber\\
    & \qquad \qquad \pi\left(D^{Y^\prime}_{Ht} \mid D^{X^\prime}_{Ht}, H^{Y^k}_{t - 1}\right)\pi\left(D^{Y^\prime}_{It} \mid D^{X^\prime}_{It}, I^{Y^k}_{1(t - 1)}\right).\label{eq:margy1}
\end{align}
Here we have a finite sample space over which to marginalise the remaining terms, since $D^{Y^\prime}_{Ht} = 0, \dots, H^{Y^k}_{t - 1}$ and $D^{Y^\prime}_{It} = 0, \dots, I^{Y^k}_{1(t - 1)}$.

Now if we consider the nested sum in (\ref{eq:sumxy}):
\begin{align}
    \sum_{\bx_t} \left[\pi\left(\bx_t \mid \by^k_{t - 1} \right)\sum_{\by_t} \pi\left(\bz_t \mid \by_t\right)\pi\left(\by_t \mid \bx_t, \by^k_{t - 1}\right)\right],\label{eq:margxy}
\end{align}
where $\pi\left(\bx_t \mid \by^k_{t - 1} \right)$ is given by (\ref{eq:simlike}) and the internal sum by (\ref{eq:margy1}). In (\ref{eq:simlike}), we can set
\begin{equation}
    \pi\left(D^{X^\prime}_{Ht}, R^{X^\prime}_{Ht} \mid H^{Y^k}_{t - 1}\right) = \pi\left(D^{X^\prime}_{Ht} \mid H^{Y^k}_{t - 1}\right)\pi\left(R^{X^\prime}_{Ht} \mid D^{X^\prime}_{Ht}, H^{Y^k}_{t - 1}\right),\label{eq:decomp1}
\end{equation}
and
\begin{equation}
    \pi\left(H^{X^\prime}_t, I^{X^\prime}_{2t}, D^{X^\prime}_{It} \mid I^{Y^k}_{t - 1}\right) = \pi\left(D^{X^\prime}_{It} \mid I^{Y^k}_{t - 1}\right)\pi\left(H^{X^\prime}_t, I^{X^\prime}_{2t} \mid D^{X^\prime}_{It}, I^{Y^k}_{t - 1}\right).\label{eq:decomp2}
\end{equation}
Since $\pi\left(D^{X^\prime}_{Ht}, R^{X^\prime}_{Ht} \mid H^{Y^k}_{t - 1}\right)$ is multinomial, both the marginal, $\pi\left(D^{X^\prime}_{Ht} \mid H^{Y^k}_{t - 1}\right)$, and conditional, $\pi\left(R^{X^\prime}_{Ht} \mid D^{X^\prime}_{Ht}, H^{Y^k}_{t - 1}\right)$, are binomial. Similarly the marginal, $\pi\left(D^{X^\prime}_{It} \mid I^{Y^k}_{t - 1}\right)$, is binomial; and the conditional, $\pi\left(H^{X^\prime}_t, I^{X^\prime}_{2t} \mid D^{X^\prime}_{It}, I^{Y^k}_{t - 1}\right)$, is multinomial. Under these decompositions, (\ref{eq:margxy}) becomes:
\begin{align}
   & \sum_{\bx_t} \left[\pi\left(\bx_t \mid \by^k_{t - 1} \right)\sum_{\by_t} \pi\left(\bz_t \mid \by_t\right)\pi\left(\by_t \mid \bx_t, \by^k_{t - 1}\right)\right] \nonumber\\
   &= \sum_{D^{X^\prime}_{Ht}}\sum_{D^{X^\prime}_{It}}\dots\sum_{E^{X^\prime}_t} \Bigg[\pi\left(\bx_t \mid \by^k_{t - 1} \right)\nonumber\\
   & \qquad \qquad \times \sum_{D^{Y^\prime}_{Ht}}\sum_{D^{Y^\prime}_{It}} \pi\left(D^{Z^\prime}_{Ht} \mid D^{Y^\prime}_{Ht}\right)\pi\left(D^{Z^\prime}_{It} \mid D^{Y^\prime}_{It}\right) \pi\left(D^{Y^\prime}_{Ht} \mid D^{X^\prime}_{Ht}, H^{Y^k}_{t - 1}\right)\pi\left(D^{Y^\prime}_{It} \mid D^{X^\prime}_{It}, I^{Y^k}_{1(t - 1)}\right)\Bigg]\nonumber\\
    &= \sum_{D^{X^\prime}_{Ht}}\sum_{D^{X^\prime}_{It}} \Bigg[\pi\left(D^{X^\prime}_{Ht} \mid H^{Y^k}_{t - 1}\right)\pi\left(D^{X^\prime}_{It} \mid I^{Y^k}_{1(t - 1)}\right) \nonumber\\
    & \qquad \qquad \times  \sum_{D^{Y^\prime}_{Ht}}\sum_{D^{Y^\prime}_{It}} \pi\left(D^{Z^\prime}_{Ht} \mid D^{Y^\prime}_{Ht}\right)\pi\left(D^{Z^\prime}_{It} \mid D^{Y^\prime}_{It}\right) \pi\left(D^{Y^\prime}_{Ht} \mid D^{X^\prime}_{Ht}, H^{Y^k}_{t - 1}\right)\pi\left(D^{Y^\prime}_{It} \mid D^{X^\prime}_{It}, I^{Y^k}_{1(t - 1)}\right)\Bigg],\label{eq:qmargxy}
\end{align}
since many of the terms in (\ref{eq:simlike}) marginalise to one in (\ref{eq:qmargxy}). Again, the sums in (\ref{eq:qmargxy}) are finite since $D^{X^\prime}_{Ht} = 0, \dots, H^{Y^k}_{t - 1}$ and $D^{X^\prime}_{It} = 0, \dots, I^{Y^k}_{1(t - 1)}$. This allows us to sample a particle index $k \sim q\left(k \mid \bz_t, \bA_{t - 1}\right)$ as required in Step \ref{apf:k} of the APF. 

In Step \ref{apf:x}, we then have to simulate $\bx^j_t \sim q\left(\bx_t \mid k^j, \bz_t, \bA_{t - 1}\right)$. This means we have to simulate from a distribution with p.m.f.
\begin{equation}
    q\left(\bx_t \mid k, \bz_t, \bA_{t - 1}\right) \propto \pi\left(\bx_t \mid \by^k_{t - 1} \right)\sum_{\by_t} \pi\left(\bz_t \mid \by_t\right)\pi\left(\by_t \mid \bx_t, \by^k_{t - 1}\right).\label{eq:qmarg}
\end{equation}
We describe below a method to do this by by simulating $D^{X^\prime}_{Ht}$ and $D^{X^\prime}_{It}$ conditionally on the data, and then simulating the rest of the states conditionally on $D^{X^\prime}_{Ht}$ and $D^{X^\prime}_{It}$ using the simulator directly (but with modified inputs).

We note that it is possible to factorise the simulator likelihood such that
\begin{align}
     \pi\left(\bx_t \mid \by^k_{t - 1} \right) &= \pi\left(D^{X^\prime}_{Ht}, D^{X^\prime}_{It} \mid \by^k_{t - 1}\right)\pi\left(\bx_t \setminus \left\{D^{X^\prime}_{Ht}, D^{X^\prime}_{It}\right\} \mid D^{X^\prime}_{Ht}, D^{X^\prime}_{It}, \by^k_{t - 1}\right),\label{eq:likefact}
\end{align}
where $\bx_t \setminus \left\{D^{X^\prime}_{Ht}, D^{X^\prime}_{It}\right\}$ denotes all states with $D^{X^\prime}_{Ht}$ and $D^{X^\prime}_{It}$ removed. We therefore set up an importance distribution
\begin{equation}
    q\left(\bx_t \mid k, \bz_t, \bA_{t - 1}\right) = q\left(D^{X^\prime}_{Ht}, D^{X^\prime}_{It} \mid \by^k_{t - 1}\right)q\left(\bx_t \setminus \left\{D^{X^\prime}_{Ht}, D^{X^\prime}_{It}\right\} \mid D^{X^\prime}_{Ht}, D^{X^\prime}_{It}, \by^k_{t - 1}\right),
\end{equation}
where $q\left(D^{X^\prime}_{Ht}, D^{X^\prime}_{It} \mid \by^k_{t - 1}\right)$ is derived by marginalising out the unwanted terms in (\ref{eq:qmarg}) using the relationship in (\ref{eq:likefact}) to give
\begin{align}
     q\left(D^{X^\prime}_{Ht}, D^{X^\prime}_{It} \mid \by^k_{t - 1}\right) &\propto \pi\left(D^{X^\prime}_{Ht} \mid H^{Y^k}_{t - 1}\right)\pi\left(D^{X^\prime}_{It} \mid I^{Y^k}_{1(t - 1)}\right) \nonumber\\
    & \qquad \qquad \times \Bigg[\sum_{D^{Y^\prime}_{Ht}}\sum_{D^{Y^\prime}_{It}} \pi\left(D^{Z^\prime}_{Ht} \mid D^{Y^\prime}_{Ht}\right)\pi\left(D^{Z^\prime}_{It} \mid D^{Y^\prime}_{It}\right)\nonumber\\
    & \qquad \qquad \qquad \qquad \times\pi\left(D^{Y^\prime}_{Ht} \mid D^{X^\prime}_{Ht}, H^{Y^k}_{t - 1}\right)\pi\left(D^{Y^\prime}_{It} \mid D^{X^\prime}_{It}, I^{Y^k}_{1(t - 1)}\right)\Bigg].\label{eq:qdx}
\end{align}
Since the ranges $D^{X^\prime}_{Ht} = 0, \dots, H^{Y^k}_t$ and $D^{X^\prime}_{It} = 0, \dots, I^{Y^k}_{1(t - 1)}$ are finite, (\ref{eq:qdx}) can be normalised and sampled from using inverse transform sampling.

Since the sum in (\ref{eq:qmarg}) only depends on $D^{X^\prime}_{Ht}$ and $D^{X^\prime}_{It}$, we can therefore choose
\begin{equation}
    q\left(\bx_t \setminus \left\{D^{X^\prime}_{Ht}, D^{X^\prime}_{It}\right\} \mid D^{X^\prime}_{Ht}, D^{X^\prime}_{It}, \by^k_{t - 1}\right) = \pi\left(\bx_t \setminus \left\{D^{X^\prime}_{Ht}, D^{X^\prime}_{It}\right\} \mid D^{X^\prime}_{Ht}, D^{X^\prime}_{It}, \by^k_{t - 1}\right).\label{eq:condq}
\end{equation}
It turns out that we can sample from $\pi\left(\bx_t \setminus \left\{D^{X^\prime}_{Ht}, D^{X^\prime}_{It}\right\} \mid D^{X^\prime}_{Ht}, D^{X^\prime}_{It}, \by^k_{t - 1}\right)$ \textit{directly from the simulator}, by adjusting some parameters and inputs slightly.

If we first consider 
\begin{equation}
    \left(D^{X^\prime}_{Ht}, R^{X^\prime}_{Ht} \mid H^{Y^k}_{t - 1}\right) \sim \mbox{Multinomial}\left(H^{Y^k}_{t - 1}, \bp\right),
\end{equation}
where $\bp = \left(p_Hp_{HD}, p_H\left(1 - p_{HD}\right), 1 - p_H\right)$. From this we have that
\begin{equation}
    \left(R^{X^\prime}_{Ht} \mid D^{X^\prime}_{Ht}, H^{Y^k}_{t - 1}\right) \sim \mbox{Binomial}\left(H^{Y^k}_{t - 1} - D^{X^\prime}_{Ht}, \frac{p_H\left(1 - p_{HD}\right)}{1 - p_Hp_{HD}}\right).\label{eq:dhbin}
\end{equation}
Therefore (\ref{eq:dhbin}) is equivalent to sampling
\begin{equation}
    \left(D^{\ast X^\prime}_{Ht}, R^{X^\prime}_{Ht} \mid H^{Y^k}_{t - 1}\right) \sim \mbox{Multinomial}\left(H^{Y^k}_{t - 1} - D^{X^\prime}_{Ht}, \bp^\ast\right),
\end{equation}
where $\bp^\ast = \left(0, \frac{p_H\left(1 - p_{HD}\right)}{1 - p_Hp_{HD}}, \frac{1 - p_H}{1 - p_Hp_{HD}}\right)$. Similarly, if we consider
\begin{equation}
    \left(H^{X^\prime}_t, I^{X^\prime}_{2t}, D^{X^\prime}_{It} \mid I^{Y^k}_{1(t - 1)}\right) \sim \mbox{Multinomial}\left(I^{Y^k}_{1(t - 1)}, \bq\right),
\end{equation}
where $\bq = \left(p_{I_1}p_{I_1H}, p_{I_1}\left(1 - p_{I_1H} - p_{I_1D}\right), p_{I_1}p_{I_1D}, 1 - p_{I_1}\right)$, then we can derive the conditional:
\begin{equation}
    \left(H^{X^\prime}_t, I^{X^\prime}_{2t} \mid D^{X^\prime}_{It}, I^{Y^k}_{1(t - 1)}\right) \sim \mbox{Multinomial}\left(I^{Y^k}_{1(t - 1)} - D^{X^\prime}_{It}, \bq^\prime\right),\label{eq:dimult}
\end{equation}
where $\bq^\prime = \left(\frac{p_{I_1}p_{I_1H}}{1 - p_{I_1}p_{I_1D}}, \frac{p_{I_1}\left(1 - p_{I_1H} - p_{I_1D}\right)}{1 - p_{I_1}p_{I_1D}}, \frac{1 - p_{I_1}}{1 - p_{I_1}p_{I_1D}}\right)$. Therefore (\ref{eq:dimult}) is equivalent to sampling
\begin{equation}
    \left(H^{X^\prime}_t, I^{X^\prime}_{2t}, D^{\ast X^\prime}_{It} \mid I^{Y^k}_{1(t - 1)} - D^{X^\prime}_{It}\right) \sim \mbox{Multinomial}\left(I^{Y^k}_{1(t - 1)} - D^{X^\prime}_{It}, \bq^\ast\right),
\end{equation}
where $\bq^\ast = \left(\frac{p_{I_1}p_{I_1H}}{1 - p_{I_1}p_{I_1D}}, \frac{p_{I_1}\left(1 - p_{I_1H} - p_{I_1D}\right)}{1 - p_{I_1}p_{I_1D}}, 0, \frac{1 - p_{I_1}}{1 - p_{I_1}p_{I_1D}}\right)$. Hence we can use the simulator directly by slightly adjusting the inputs and parameters to sample from the correct conditional distribution (\ref{eq:condq}). This also means that most of the complex parts of the simulator likelihood cancel in the importance weights.

Finally, to sample from (\ref{eq:sum}) we note the factorisation
\begin{align}
    q\left(\by_t \mid \bx_t, k, \bz_t, \bA_{t - 1}\right) &= q\left(D^{Y^\prime}_{Ht}, D^{Y^\prime}_{It} \mid \bx_t, k, \bz_t, \bA_{t - 1}\right)q\left(\by_t \setminus \left\{D^{Y^\prime}_{Ht}, D^{Y^\prime}_{It}\right\} \mid \bx_t, k, \bz_t, \bA_{t - 1}\right)
\end{align}
where
\begin{align}
    q\left(D^{Y^\prime}_{Ht}, D^{Y^\prime}_{It} \mid \bx_t, k, \bz_t, \bA_{t - 1}\right) &\propto \pi\left(D^{Z^\prime}_{Ht} \mid D^{Y^\prime}_{Ht}\right)\pi\left(D^{Z^\prime}_{It} \mid D^{Y^\prime}_{It}\right)\pi\left(D^{Y^\prime}_{Ht} \mid D^{X^\prime}_{Ht}, H^{Y^k}_{t - 1}\right)\nonumber\\
    & \qquad \qquad \times \pi\left(D^{Y^\prime}_{It} \mid D^{X^\prime}_{It}, I^{Y^k}_{1(t - 1)}\right) \qquad \mbox{from (\ref{eq:mdlike}) and (\ref{eq:obsdens})},
\end{align}
where $D^{Y^\prime}_{Ht} = 0, \dots, H^{Y^k}_{t - 1}$ and $D^{Y^\prime}_{It} = 0, \dots, I^{Y^k}_{1(t - 1)}$. Then
\begin{align}
    q\left(\by_t \setminus \left\{D^{Y^\prime}_{Ht}, D^{Y^\prime}_{It}\right\} \mid \bx_t, k, \bz_t, \bA_{t - 1}\right) &= \pi\left(\by_t \setminus \left\{D^{Y^\prime}_{Ht}, D^{Y^\prime}_{It}\right\} \mid \bx_t, k, \bz_t, \bA_{t - 1}\right)
\end{align}
which is straightforward to derive from (\ref{eq:mdlike}).

\subsection{Unbiasedness of likelihood estimator}

This maintains the sufficient condition for unbiasedness of the marginal likelihood estimate, since
\begin{align*}
    E\left[\hat{\pi}\left(z_t \mid A_{t - 1}\right)\right] &= E\left[\left(\frac{1}{M}\sum_{j = 1}^M w^j_t\right)\left(\sum_{j = 1}^M w^j_{t - 1 \mid t}\right)\right]\\
    &= \left(\sum_{j = 1}^M w^j_{t - 1 \mid t}\right)E\left[\frac{1}{M}\sum_{j = 1}^M w^j_t\right] \qquad \mbox{since $w^j_{t - 1 \mid t}$ known given $A_{t - 1}$}\\
    &= \left(\sum_{j = 1}^M w^j_{t - 1 \mid t}\right)\left(\frac{1}{M}\sum_{j = 1}^M E\left[w^j_t\right]\right) \qquad \mbox{from linearity of expectations}\\
    &= \frac{\left(\sum_{j = 1}^M w^j_{t - 1 \mid t}\right)}{M}\sum_{j = 1}^M \\
    & \qquad \sum_{k = 1}^M\int_{y_t} \int_{x_t}  \frac{\pi\left(z_t \mid y_t\right)\pi\left(y_t \mid x_t\right)\pi\left(x_t \mid y^{k}_{t - 1}\right)}{q\left(z_t \mid y^{k}_{t - 1}\right)q\left(y_t \mid x_t, k, z_t, A_{t - 1}\right)q\left(x_t \mid k, z_t, A_{t - 1}\right)}q\left(y_t, x_t, k \mid z_t, A_{t - 1}\right)\mathrm{d}x_t\mathrm{d}y_t\\
    &= \frac{\left(\sum_{j = 1}^M w^j_{t - 1 \mid t}\right)}{M}\sum_{j = 1}^M \\
    & \qquad \sum_{k = 1}^M\int_{y_t} \int_{x_t}  \frac{\pi\left(z_t \mid y_t\right)\pi\left(y_t \mid x_t\right)\pi\left(x_t \mid y^{k}_{t - 1}\right)}{q\left(z_t \mid y^{k}_{t - 1}\right)}q\left(k \mid z_t, A_{t - 1}\right)\mathrm{d}x_t\mathrm{d}y_t \qquad \mbox{from (\ref{eq:qcond})}\\
    &= \frac{\left(\sum_{j = 1}^M w^j_{t - 1 \mid t}\right)}{M}\sum_{j = 1}^M \\
    & \qquad \sum_{k = 1}^M\int_{y_t} \int_{x_t}  \frac{\pi\left(z_t \mid y_t\right)\pi\left(y_t \mid x_t\right)\pi\left(x_t \mid y^{k}_{t - 1}\right)}{q\left(z_t \mid y^{k}_{t - 1}\right)}\left(\frac{w^k_{t - 1 \mid t}}{\sum_{j = 1}^M w^j_{t - 1 \mid t}}\right)\mathrm{d}x_t\mathrm{d}y_t\\
    &= \frac{1}{M}\sum_{j = 1}^M \sum_{k = 1}^M\int_{y_t} \int_{x_t}  \frac{\pi\left(z_t \mid y_t\right)\pi\left(y_t \mid x_t\right)\pi\left(x_t \mid y^{k}_{t - 1}\right)}{q\left(z_t \mid y^{k}_{t - 1}\right)}\cdot\pi^k_{t - 1} q\left(z_t \mid y^{k}_{t - 1}\right)\mathrm{d}x_t\mathrm{d}y_t\\
    &= \frac{1}{M}\sum_{j = 1}^M\sum_{k = 1}^M \pi^k_{t - 1} \int_{y_t} \int_{x_t} \pi\left(z_t \mid y_t\right)\pi\left(y_t \mid x_t\right)\pi\left(x_t \mid y^k_{t - 1}\right)\mathrm{d}x_t\mathrm{d}y_t\\
    &= \frac{1}{M}\sum_{j = 1}^M\sum_{k = 1}^M \pi\left(z_t \mid y^k_{t - 1}\right)\pi^k_{t - 1}\\
    &= \sum_{k = 1}^M \pi\left(z_t \mid y^k_{t - 1}\right)\pi^k_{t - 1}.
\end{align*}
